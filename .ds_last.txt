import type { Settings, WeatherBundle } from "../types";

const AQI_STOPS: Array<[number, string]> = [
  [0, "#4ade80"],
  [50, "#facc15"],
  [100, "#fb923c"],
  [150, "#ef4444"],
  [200, "#a855f7"],
  [300, "#a855f7"],
];

const UV_STOPS: Array<[number, string]> = [
  [0, "#4ade80"],
  [2, "#4ade80"],
  [3, "#facc15"],
  [5, "#facc15"],
  [6, "#fb923c"],
  [7, "#fb923c"],
  [8, "#ef4444"],
  [10, "#ef4444"],
  [11, "#a855f7"],
];

function rampGradient(stops: Array<[number, string]>, max: number): string {
  const parts = stops.map(([v, c]) => `${c} ${Math.round((v / max) * 100)}%`);
  return `linear-gradient(90deg, ${parts.join(", ")})`;
}

function makeCard(title: string, iconName: Parameters<typeof icon>[0], extra = ""): HTMLElement {
  const card = document.createElement("section");
  card.className = `card info${extra ? " " + extra : ""}`;
  const h = document.createElement("h2");
  h.className = "card-title";
  h.innerHTML = `${icon(iconName, 13)}<span>${esc(title)}</span>`;
  card.appendChild(h);
  return card;
}

function aqiCategory(aqi: number): string {
  if (aqi <= 50) return "Good";
  if (aqi <= 100) return "Moderate";
  if (aqi <= 150) return "Unhealthy for sensitive";
  if (aqi <= 200) return "Unhealthy";
  if (aqi <= 300) return "Very unhealthy";
  return "Hazardous";
}

function aqiSentence(aqi: number): string {
  if (aqi <= 50) return "Air quality is satisfactory today.";
  if (aqi <= 100) return "Air quality is acceptable; sensitive people should take care.";
  if (aqi <= 150) return "Sensitive groups may feel effects from the air.";
  if (aqi <= 200) return "Everyone may begin to feel effects from the air.";
  if (aqi <= 300) return "Health alert: air is very unhealthy.";
  return "Health warning: hazardous air.";
}

function uvCategory(uv: number): string {
  if (uv <= 2) return "Low";
  if (uv <= 5) return "Moderate";
  if (uv <= 7) return "High";
  if (uv <= 10) return "Very high";
  return "Extreme";
}

function rampBar(gradient: string, pos: number): string {
  const pct = Math.max(0, Math.min(100, pos * 100));
  return `<div class="ramp"><div class="ramp-fill" style="background:${gradient}"></div><div class="ramp-dot" style="left:${pct}%"></div></div>`;
}

function airCard(b: WeatherBundle): HTMLElement {
  const aqi = b.air!.usAqi;
  const card = makeCard("Air Quality", "leaf", "span-2");
  const body = document.createElement("div");
  body.className = "info-body";
  body.innerHTML = `
    <div class="value">${aqi}</div>

    <div class="sentence">${esc(aqiSentence(aqi))}</div>
    ${rampBar(rampGradient(AQI_STOPS, 300), Math.min(aqi, 300) / 300)}
  `;
  card.appendChild(body);

}

function uvCard(b: WeatherBundle): HTMLElement {
  const uv = Math.round(b.current.uvIndex);
  const peak = Math.round(b.daily[0]?.uvIndexMax ?? uv);
  const card = makeCard("UV Index", "uv");
  const body = document.createElement("div");
  body.className = "info-body";
  body.innerHTML = `
    <div class="value">${uv}</div>
    <div class="sub">${esc(uvCategory(uv))}</div>
    <div class="sentence">Peak today: ${peak}</div>
    ${rampBar(rampGradient(UV_STOPS, 11), Math.min(uv, 11) / 11)}
  `;
  card.appendChild(body);
  return card;
}

function timeParts(iso: string): number {
  const t = iso.length > 10 ? iso.slice(11, 16) : "00:00";
  const [h, m] = t.split(":").map((x) => parseInt(x, 10));
  return (Number.isFinite(h) ? h : 0) * 60 + (Number.isFinite(m) ? m : 0);
}

function sunCard(b: WeatherBundle, s: Settings): HTMLElement {
  const day = b.daily[0];
  const card = makeCard("Sun", "sun");
  const body = document.createElement("div");
  body.className = "info-body";
  const rise = day ? timeParts(day.sunrise) : 0;
  const set = day ? timeParts(day.sunset) : 0;
  const now = timeParts(b.current.time);
  let t = (now - rise) / (set - rise || 1);
  const dimmed = t < 0 || t > 1;
  t = Math.max(0, Math.min(1, t));

  const w = 180;
  const h = 60;
  const cx = w * t;
  const cy = h - Math.sin(Math.PI * t) * (h - 8);
  const dot = dimmed ? `style="opacity:.4"` : "";

  body.innerHTML = `
    <div class="sun-times">
      <span>${icon("sunrise", 14)} ${esc(day ? clock(day.sunrise, s) : "--")}</span>
      <span>${icon("sunset", 14)} ${esc(day ? clock(day.sunset, s) : "--")}</span>
    </div>
    <svg class="sun-arc" viewBox="0 0 ${w} ${h}" width="100%" height="${h}" aria-hidden="true">
      <path d="M0 ${h - 1} Q ${w / 2} ${-h + 12} ${w} ${h - 1}" fill="none" stroke="rgba(255,255,255,.35)" stroke-width="1.5" stroke-dasharray="4 4"/>
      <line x1="0" y1="${h}" x2="${w}" y2="${h}" stroke="rgba(255,255,255,.28)" stroke-width="1"/>
      <circle cx="${cx.toFixed(1)}" cy="${cy.toFixed(1)}" r="4.5" fill="#fff" ${dot}/>
    </svg>
  `;
  card.appendChild(body);
  return card;
}

function windCard(b: WeatherBundle, s: Settings): HTMLElement {
  const card = makeCard("Wind", "wind");
  const body = document.createElement("div");
  body.className = "info-body wind";
  const dir = b.current.windDirection;
  const deg = ((dir % 360) + 360) % 360;
  const speed = b.current.windSpeed;
  const wv = unitOf(s);
  const converted = Math.round(wv.conv(speed));

  const cx = 60;
  const cy = 60;
  let ticks = "";
  for (let i = 0; i < 12; i++) {
    const a = (i * 30 * Math.PI) / 180;
    const inner = i % 3 === 0 ? 48 : 52;
    const x1 = cx + Math.sin(a) * inner;
    const y1 = cy - Math.cos(a) * inner;
    const x2 = cx + Math.sin(a) * 56;
    const y2 = cy - Math.cos(a) * 56;
    ticks += `<line x1="${x1.toFixed(1)}" y1="${y1.toFixed(1)}" x2="${x2.toFixed(1)}" y2="${y2.toFixed(1)}" stroke="rgba(255,255,255,.4)" stroke-width="${i % 3 === 0 ? 1.6 : 1}"/>`;
  }

  const letters = [
    { t: "N", x: cx, y: 12 },
    { t: "E", x: 112, y: cy + 4 },
    { t: "S", x: cx, y: 116 },
    { t: "W", x: 8, y: cy + 4 },
  ]
    .map((l) => `<text x="${l.x}" y="${l.y}" font-size="10" fill="rgba(255,255,255,.7)" text-anchor="middle" font-family="inherit">${l.t}</text>`)
    .join("");

  body.innerHTML = `
    <div class="wind-compass">
      <svg viewBox="0 0 120 120" width="120" height="120" aria-hidden="true">
        <circle cx="${cx}" cy="${cy}" r="58" fill="none" stroke="rgba(255,255,255,.18)" stroke-width="1"/>
        ${ticks}
        ${letters}
        <g transform="rotate(${deg} 60 60)">
          <path d="M60 30 L66 46 L60 42 L54 46 Z" fill="#fff"/>
          <line x1="60" y1="42" x2="60" y2="64" stroke="rgba(255,255,255,.55)" stroke-width="1.6"/>
        </g>
        <circle cx="${cx}" cy="${cy}" r="2" fill="#fff"/>
      </svg>
    </div>
    <div class="value">${converted}<span class="unit">${esc(wv.unit)}</span></div>
    <div class="sub">${esc(compass(deg))}</div>
  `;
  card.appendChild(body);
  return card;
}

function unitOf(s: Settings): { unit: string; conv: (kmh: number) => number } {
  if (s.wind === "mph") return { unit: "mph", conv: (k) => k * 0.621371 };
  if (s.wind === "ms") return { unit: "m/s", conv: (k) => k / 3.6 };
  return { unit: "km/h", conv: (k) => k };
}

export function renderCards(b: WeatherBundle, s: Settings): HTMLElement[] {
  const out: HTMLElement[] = [];

  out.push(uvCard(b));
  out.push(sunCard(b, s));
  out.push(windCard(b, s));
  out.push(precipCard(b));
  out.push(feelsCard(b, s));
  out.push(humidityCard(b));
  out.push(visibilityCard(b));

import type { Location, Settings, WeatherBundle } from "../types";
import { state, set, subscribe } from "../state";
import { searchCity, saveLocation, removeLocation, getWeather, getCached } from "../api";
import { esc, saveSettings, tempDeg } from "../format";
import { describe } from "../theme";
import { conditionIcon, icon } from "../icons";

const cache = new Map<string, WeatherBundle>();
const fetching = new Set<string>();

function key(loc: Location): string {
  return `${loc.latitude.toFixed(2)},${loc.longitude.toFixed(2)}`;
}

function sameLoc(a: Location, b: Location | null): boolean {
  if (!b) return false;
  return a.latitude === b.latitude && a.longitude === b.longitude;
}

function secondary(loc: Location): string {
  const parts = [loc.admin1, loc.country].filter((x): x is string => !!x);
  return parts.join(", ");
}

export function initSidebar(root: HTMLElement, onSelect: (loc: Location) => void): void {
  // Toggle button (mobile only).
  const existingToggle = document.querySelector(".sidebar-toggle");
  if (!existingToggle) {
    const toggle = document.createElement("button");
    toggle.className = "sidebar-toggle";
    toggle.setAttribute("aria-label", "Toggle sidebar");
    toggle.innerHTML = icon("menu", 20);
    toggle.addEventListener("click", () => root.classList.toggle("open"));
    document.body.appendChild(toggle);
  }

  // Backdrop for mobile.
  let backdrop: HTMLButtonElement | null = null;
  const ensureBackdrop = (): HTMLButtonElement => {
    if (!backdrop) {
      backdrop = document.createElement("button");
      backdrop.className = "sidebar-backdrop";
      backdrop.setAttribute("aria-label", "Close");
      backdrop.addEventListener("click", () => root.classList.remove("open"));
      document.body.appendChild(backdrop);
    }
    return backdrop;
  };
  ensureBackdrop();

  root.replaceChildren();

  // Search box (built once).
  const searchWrap = document.createElement("div");
  searchWrap.className = "side-search";
  searchWrap.innerHTML = `
    <span class="side-search-icon">${icon("search", 16)}</span>
    <input type="search" class="side-search-input" placeholder="Search city" autocomplete="off" spellcheck="false"/>
  `;
  const input = searchWrap.querySelector("input") as HTMLInputElement;

  const results = document.createElement("div");
  results.className = "side-results";
  results.hidden = true;

  const list = document.createElement("div");
  list.className = "side-list";

  const settingsBox = document.createElement("div");
  settingsBox.className = "side-settings";

  root.append(searchWrap, results, list, settingsBox);

  let reqId = 0;
  let timer: number | undefined;

  input.addEventListener("input", () => {
    if (timer !== undefined) window.clearTimeout(timer);
    const q = input.value.trim();
    if (!q) {
      results.hidden = true;
      results.replaceChildren();
      return;
    }
    timer = window.setTimeout(() => void runSearch(q), 250);
  });

  input.addEventListener("keydown", (e) => {
    if (e.key === "Escape") {
      input.value = "";
      results.hidden = true;
      results.replaceChildren();
      input.blur();
    }
  });

  document.addEventListener("click", (e) => {
    if (!searchWrap.contains(e.target as Node) && !results.contains(e.target as Node)) {
      results.hidden = true;
    }
  });

  const runSearch = async (q: string): Promise<void> => {
    const id = ++reqId;
    try {
      const found = await searchCity(q);
      if (id !== reqId) return;
      renderResults(found);
    } catch {
      if (id !== reqId) return;
      renderResults([]);
    }
  };

  const renderResults = (found: Location[]): void => {
    results.replaceChildren();
    results.hidden = false;
    if (found.length === 0) {
      const empty = document.createElement("div");
      empty.className = "side-empty";
      empty.textContent = "No matches";
      results.appendChild(empty);
      return;
    }
    for (const loc of found.slice(0, 8)) {
      const item = document.createElement("button");
      item.className = "side-result";
      item.innerHTML = `
        <span class="side-result-name">${esc(loc.name)}</span>
        <span class="side-result-sub">${esc(secondary(loc))}</span>
      `;
      item.addEventListener("click", () => {
        void (async () => {
          try {
            const updated = await saveLocation(loc);
            set({ saved: updated });
          } catch {
            /* ignore */
          }
          input.value = "";
          results.hidden = true;
          results.replaceChildren();
          input.blur();
          onSelect(loc);
        })();
      });
      results.appendChild(item);
    }
  };

  // Prefetch temps for saved cities.
  const ensureTemps = (saved: Location[]): void => {
    for (const loc of saved) {
      const k = key(loc);
      if (cache.has(k) || fetching.has(k)) continue;
      void (async () => {
        fetching.add(k);
        try {
          const cached = await getCached(loc);
          if (cached) {
            cache.set(k, cached);
            renderList();
          }
          const fresh = await getWeather(loc, false);
          cache.set(k, fresh);
          renderList();
        } catch {
          /* ignore */
        } finally {
          fetching.delete(k);
        }
      })();
    }
  };

  const renderList = (): void => {
    const saved = state.saved;
    ensureTemps(saved);
    list.replaceChildren();
    if (saved.length === 0) {
      const empty = document.createElement("div");
      empty.className = "side-empty";
      empty.textContent = "No saved cities";
      list.appendChild(empty);
      return;
    }
    for (const loc of saved) {
      const active = sameLoc(loc, state.current);
      const row = document.createElement("div");
      row.className = `city${active ? " active" : ""}`;

      const b = cache.get(key(loc));
      const cond = b ? describe(b.current.weatherCode, b.current.isDay) : null;
      const tempTxt = b ? tempDeg(b.current.temperature, state.settings) : "\u2013";

      row.innerHTML = `
        <button class="city-main" type="button">
          <span class="city-icon">${cond ? conditionIcon(cond.kind, b!.current.isDay, 20) : ""}</span>
          <span class="city-text">
            <span class="city-name">${esc(loc.name)}</span>
            <span class="city-sub">${esc(secondary(loc))}</span>
          </span>
          <span class="city-temp">${esc(tempTxt)}</span>
        </button>
        <button class="city-remove" type="button" aria-label="Remove">${icon("close", 12)}</button>
      `;

      const mainBtn = row.querySelector(".city-main") as HTMLButtonElement;
      mainBtn.addEventListener("click", () => {
        root.classList.remove("open");
        onSelect(loc);
      });

      const removeBtn = row.querySelector(".city-remove") as HTMLButtonElement;
      removeBtn.addEventListener("click", (e) => {
        e.stopPropagation();
        void (async () => {
          try {
            const updated = await removeLocation(loc.latitude, loc.longitude);
            set({ saved: updated });
            if (sameLoc(loc, state.current) && updated.length > 0) {
              onSelect(updated[0]);
            }
          } catch {
            /* ignore */
          }
        })();
      });

      list.appendChild(row);
    }
  };

  const renderSettings = (): void => {
    const s = state.settings;
    settingsBox.innerHTML = `<div class="side-section-title">SETTINGS</div>`;
    settingsBox.appendChild(segmented([
      { label: "\u00B0C", value: "c", active: s.temp === "c" },
      { label: "\u00B0F", value: "f", active: s.temp === "f" },
    ], (v) => update({ ...state.settings, temp: v as Settings["temp"] })));
    settingsBox.appendChild(segmented([
      { label: "km/h", value: "kmh", active: s.wind === "kmh" },
      { label: "mph", value: "mph", active: s.wind === "mph" },
      { label: "m/s", value: "ms", active: s.wind === "ms" },
    ], (v) => update({ ...state.settings, wind: v as Settings["wind"] })));
    settingsBox.appendChild(segmented([
      { label: "24h", value: "24", active: s.clock === "24" },
      { label: "12h", value: "12", active: s.clock === "12" },
    ], (v) => update({ ...state.settings, clock: v as Settings["clock"] })));
  };

  interface Seg { label: string; value: string; active: boolean; }
  const segmented = (items: Seg[], onPick: (v: string) => void): HTMLElement => {
    const box = document.createElement("div");
    box.className = "seg";
    for (const it of items) {
      const btn = document.createElement("button");
      btn.type = "button";
      btn.className = `seg-btn${it.active ? " active" : ""}`;
      btn.textContent = it.label;
      btn.addEventListener("click", () => onPick(it.value));
      box.appendChild(btn);
    }
    return box;
  };

  let lastSettings = state.settings;
  const update = (s: Settings): void => {
    lastSettings = s;
    set({ settings: s });
    saveSettings(s);
  };
  void lastSettings;

  subscribe((s) => {
    if (s.saved !== undefined) renderList();
    if (s.settings !== lastSettings) {
      lastSettings = s.settings;
      renderSettings();
    }
  });

  renderList();
  renderSettings();
}

import "./styles.css";
import "./styles/cards.css";
import "./styles/sidebar.css";
import type { Location, WeatherBundle } from "./types";
import { describe, skyGradient } from "./theme";
import { getCached, getSaved, getWeather, saveLocation } from "./api";
import { state, set, subscribe } from "./state";
import { renderHero } from "./ui/hero";
import { renderHourly } from "./ui/hourly";
import { renderDaily } from "./ui/daily";
import { renderCards } from "./ui/cards";
import { initSidebar } from "./ui/sidebar";

const DEFAULT: Location = {
  name: "Berlin",
  country: "Germany",
  admin1: "Berlin",
  latitude: 52.52,
  longitude: 13.405,
};

let skyFront = 0;
let skyEls: HTMLDivElement[] = [];
let currentMain: HTMLElement | null = null;

function setupShell(): { main: HTMLElement } {
  const app = document.getElementById("app");
  if (!app) throw new Error("missing #app");

  const sky = document.createElement("div");
  sky.className = "sky";
  skyEls = [0, 1].map(() => {
    const l = document.createElement("div");
    l.className = "sky-layer";
    sky.appendChild(l);
    return l;
  });

  const sidebar = document.createElement("aside");
  sidebar.id = "sidebar";
  sidebar.className = "sidebar";

  const main = document.createElement("main");
  main.id = "main";

  app.replaceChildren(sky, sidebar, main);
  return { main };
}

function applySky(bundle: WeatherBundle): void {
  const c = bundle.current;
  const cond = describe(c.weatherCode, c.isDay);
  const grad = skyGradient(cond.kind, c.isDay);
  const back = skyEls[1 - skyFront];
  const front = skyEls[skyFront];
  if (front.style.backgroundImage === grad) return;

  back.style.backgroundImage = grad;
  void back.offsetWidth;
  back.classList.add("active");
  front.classList.remove("active");
  skyFront = 1 - skyFront;
}

function renderMain(main: HTMLElement, b: WeatherBundle): void {
  currentMain = main;
  main.replaceChildren(
    renderHero(b, state.settings),
    renderHourly(b, state.settings),
  );
  const grid = document.createElement("section");
  grid.id = "grid";
  grid.className = "grid";
  const daily = renderDaily(b, state.settings);
  daily.classList.add("span-2", "row-2");
  grid.appendChild(daily);
  for (const card of renderCards(b, state.settings)) grid.appendChild(card);
  main.appendChild(grid);
}

function renderPlaceholder(main: HTMLElement, text: string): void {
  const box = document.createElement("div");
  box.className = "loading";
  box.textContent = text;
  main.replaceChildren(box);
}

function renderError(main: HTMLElement, retry: () => void): void {
  const box = document.createElement("div");
  box.className = "error";
  const msg = document.createElement("div");
  msg.textContent = state.error ?? "Something went wrong.";
  const btn = document.createElement("button");
  btn.textContent = "Retry";
  btn.addEventListener("click", retry);
  box.append(msg, btn);
  main.replaceChildren(box);
}

async function selectLocation(loc: Location): Promise<void> {
  const main = currentMain ?? document.getElementById("main");
  if (!main) return;

  set({ current: loc, loading: true, error: null });

  try {
    const cached = await getCached(loc);
    if (cached) {
      set({ bundle: cached });
      applySky(cached);
      renderMain(main, cached);
    } else if (!state.bundle) {
      renderPlaceholder(main, "Loading weather\u2026");
    }
  } catch {
    /* no cache */
  }

  try {
    const b = await getWeather(loc, false);
    set({ bundle: b, loading: false });
    applySky(b);
    renderMain(main, b);
  } catch (err) {
    set({ loading: false, error: String(err) });
    renderError(main, () => void selectLocation(loc));
  }
}

async function init(): Promise<void> {
  const { main } = setupShell();
  currentMain = main;

  initSidebar(document.getElementById("sidebar")!, (loc) => {
    void selectLocation(loc);
  });

  let saved: Location[] = [];
  try {
    saved = await getSaved();
  } catch {
    saved = [];
  }
  if (saved.length === 0) {
    try {
      saved = await saveLocation(DEFAULT);
    } catch {
      saved = [DEFAULT];
    }
  }
  set({ saved });
  const loc = saved[0] ?? DEFAULT;

  try {
    const cached = await getCached(loc);
    if (cached) {
      set({ bundle: cached, current: loc });
      applySky(cached);
      renderMain(main, cached);
    }
  } catch {
    /* no cache */
  }

  set({ current: loc, loading: true, error: null });
  try {
    const b = await getWeather(loc, false);
    set({ bundle: b, loading: false });
    applySky(b);
    renderMain(main, b);
  } catch (err) {
    set({ loading: false, error: String(err) });
    renderError(main, () => void selectLocation(loc));
  }

  const refresh = () => {
    const cur = state.current;
    if (!cur || state.loading) return;
    void (async () => {
      try {
        const b = await getWeather(cur, false);
        set({ bundle: b });
        applySky(b);
        renderMain(main, b);
      } catch {
        /* keep last bundle */
      }
    })();
  };

  // Re-render on unit changes without refetching.
  subscribe((s) => {
    if (s.bundle && s.settings) {
      const m = currentMain;
      if (m && m.querySelector("#grid")) renderMain(m, s.bundle);
    }
  });

  setInterval(refresh, 10 * 60 * 1000);
  document.addEventListener("visibilitychange", () => {
    if (document.visibilityState === "visible") refresh();
  });
}

void init();
