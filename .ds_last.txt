import type { Settings } from "./types";

const KEY = "weather.settings";

export const DEFAULT_SETTINGS: Settings = { temp: "c", wind: "kmh", clock: "24" };

export function loadSettings(): Settings {
  try {
    const raw = localStorage.getItem(KEY);
    if (!raw) return { ...DEFAULT_SETTINGS };
    const parsed = JSON.parse(raw) as Partial<Settings>;
    return { ...DEFAULT_SETTINGS, ...parsed };
  } catch {
    return { ...DEFAULT_SETTINGS };
  }
}

export function saveSettings(s: Settings): void {
  try {
    localStorage.setItem(KEY, JSON.stringify(s));
  } catch {
    /* ignore storage errors */
  }
}

export function esc(s: string): string {
  return s
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#39;");
}

function toF(c: number): number {
  return c * 9 / 5 + 32;
}

export function temp(c: number, s: Settings): string {
  const v = s.temp === "f" ? toF(c) : c;
  return String(Math.round(v));
}

export function tempDeg(c: number, s: Settings): string {
  return `${temp(c, s)}\u00B0`;
}

export function wind(kmh: number, s: Settings): { value: string; unit: string } {
  if (s.wind === "mph") return { value: String(Math.round(kmh * 0.621371)), unit: "mph" };
  if (s.wind === "ms") return { value: String(Math.round(kmh / 3.6)), unit: "m/s" };
  return { value: String(Math.round(kmh)), unit: "km/h" };
}

interface ClockParts {
  h: number;
  m: number;
}

function parseLocal(iso: string): ClockParts {
  // ISO strings from backend are local to the location; take the wall-clock parts.
  const t = iso.length > 10 ? iso.slice(11, 16) : "00:00";
  const [hh, mm] = t.split(":").map((x) => parseInt(x, 10));
  return { h: Number.isFinite(hh) ? hh : 0, m: Number.isFinite(mm) ? mm : 0 };
}

function hour12(h: number, m: number): string {
  const period = h < 12 ? "AM" : "PM";
  const h12 = h % 12 === 0 ? 12 : h % 12;
  const hasMin = m > 0;
  const base = hasMin ? `${h12}:${String(m).padStart(2, "0")}` : String(h12);
  return `${base} ${period}`;
}

export function hourLabel(iso: string, s: Settings): string {
  const { h } = parseLocal(iso);
  if (s.clock === "12") return hour12(h, 0);
  return String(h);
}

export function clock(iso: string, s: Settings): string {
  const { h, m } = parseLocal(iso);
  if (s.clock === "12") return hour12(h, m);
  return `${String(h).padStart(2, "0")}:${String(m).padStart(2, "0")}`;
}

const WEEKDAYS = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];

export function dayName(isoDate: string, index: number): string {
  if (index === 0) return "Today";
  const [y, mo, d] = isoDate.split("-").map((x) => parseInt(x, 10));
  const date = new Date(y, (mo || 1) - 1, d || 1);
  return WEEKDAYS[date.getDay()];
}

export function compass(deg: number): string {
  const dirs = ["N", "NE", "E", "SE", "S", "SW", "W", "NW"];
  const i = Math.round(((deg % 360) / 45)) % 8;
  return dirs[(i + 8) % 8];
}

// Temperature ramp stops (metric). Linear interpolate between them.
const RAMP: Array<[number, [number, number, number]]> = [
  [0, [124, 196, 255]],
  [10, [110, 231, 183]],
  [20, [253, 224, 71]],
  [30, [251, 146, 60]],
];

export function tempColor(c: number): string {
  if (c <= RAMP[0][0]) return rgb(RAMP[0][1]);
  const last = RAMP[RAMP.length - 1];
  if (c >= last[0]) return rgb(last[1]);
  for (let i = 0; i < RAMP.length - 1; i++) {
    const [a, ca] = RAMP[i];
    const [b, cb] = RAMP[i + 1];
    if (c >= a && c <= b) {
      const t = (c - a) / (b - a);
      return rgb([
        Math.round(ca[0] + (cb[0] - ca[0]) * t),
        Math.round(ca[1] + (cb[1] - ca[1]) * t),
        Math.round(ca[2] + (cb[2] - ca[2]) * t),
      ]);
    }
  }
  return rgb(last[1]);
}

function rgb([r, g, b]: [number, number, number]): string {
  return `rgb(${r}, ${g}, ${b})`;
}

  bundle: null,
  loading: false,
  error: null,
};

type Listener = (s: State) => void;
const listeners: Listener[] = [];

export function subscribe(fn: Listener): void {
  listeners.push(fn);
}

export function set(partial: Partial<State>): void {
  Object.assign(state, partial);
  for (const fn of listeners) fn(state);
}

import type { Settings, WeatherBundle } from "../types";
import { describe, skyGradient } from "../theme";
import { clock, esc, hourLabel, temp, tempDeg } from "../format";
import { conditionIcon } from "../icons";

export function renderHero(b: WeatherBundle, s: Settings): HTMLElement {
  const c = b.current;
  const cond = describe(c.weatherCode, c.isDay);
  const today = b.daily[0];

  const location = esc(b.location.name);
  const sub = [b.location.admin1, b.location.country].filter(Boolean).map((x) => esc(x as string)).join(", ");

  let summary = `Feels like ${tempDeg(c.apparent, s)}.`;
  const wetHour = b.hourly.find((h) => (h.precipitationProbability ?? 0) >= 40);
  if (wetHour) summary += ` Rain likely around ${hourLabel(wetHour.time, s)}.`;

  const stale = b.stale
    ? `<div class="pill">Offline, showing saved data</div>`
    : "";

  const hl = today
    ? `H ${temp(today.tempMax, s)}&deg;&nbsp;&nbsp;L ${temp(today.tempMin, s)}&deg;`
    : "";

  const wrap = document.createElement("section");
  wrap.className = "hero";
  wrap.dataset.sky = skyGradient(cond.kind, c.isDay);
  wrap.innerHTML = `
    <div class="hero-city">${location}${sub ? `<small>${sub}</small>` : ""}</div>
    <div class="hero-temp">${temp(c.temperature, s)}<sup>&deg;</sup></div>
    <div class="hero-cond">${conditionIcon(cond.kind, c.isDay, 20)} ${esc(cond.label)}</div>
    <div class="hero-hl">${hl}</div>
    <div class="hero-summary">${esc(summary)}</div>
    ${stale}
  `;
  // ensure the icon doesn't break the flow (keep it inline)
  const condEl = wrap.querySelector(".hero-cond");
  if (condEl) condEl.innerHTML = condEl.innerHTML;
  return wrap;
}

/** Exposed for potential callers that need the local time string. */

  scroll.className = "hourly-scroll";

  // Sparkline sized to content width.
  const contentW = hours.length * (COL_W + GAP) - GAP;
  const svg = sparkline(hours.map((h) => h.temperature));

  for (let i = 0; i < hours.length; i++) {
    const h = hours[i];
    const cond = describe(h.weatherCode, h.isDay);
    const col = document.createElement("div");
    col.className = "hour-col" + (i === 0 ? " is-now" : "");
    const label = i === 0 ? "Now" : hourLabel(h.time, s);
    const rain = (h.precipitationProbability ?? 0) >= 20
      ? `<div class="hour-rain">${h.precipitationProbability}%</div>`
      : `<div class="hour-rain"></div>`;
    col.innerHTML = `
      <div class="hour-label">${label}</div>
      ${conditionIcon(cond.kind, h.isDay, 22)}
      <div class="hour-temp">${temp(h.temperature, s)}&deg;</div>
      ${rain}
    `;
    scroll.appendChild(col);
  }

  // Insert sparkline absolutely positioned behind temps.
  const svgWrap = document.createElement("div");
  svgWrap.className = "hourly-spark";
  svgWrap.style.width = `${contentW}px`;
  svgWrap.style.height = `${SPARK_H}px`;
  svgWrap.innerHTML = svg;
  scroll.insertBefore(svgWrap, scroll.firstChild);

  card.appendChild(scroll);
  return card;
}

function sparkline(values: number[]): string {
  const n = values.length;
  if (n === 0) return "";
  const w = n * (COL_W + GAP) - GAP;
  const min = Math.min(...values);
  const max = Math.max(...values);
  const span = max - min || 1;
  const pad = 6;
  const h = SPARK_H;
  const pts = values.map((v, i) => {
    const x = i * (COL_W + GAP) + COL_W / 2;
    const y = pad + (1 - (v - min) / span) * (h - pad * 2);
    return [x, y] as const;
  });

  let d = `M ${pts[0][0]} ${pts[0][1]}`;
  for (let i = 0; i < n - 1; i++) {
    const [x0, y0] = pts[i];
    const [x1, y1] = pts[i + 1];
    const cx = (x0 + x1) / 2;
    d += ` C ${cx} ${y0}, ${cx} ${y1}, ${x1} ${y1}`;
  }

  const dots = pts
    .map(([x, y]) => `<circle cx="${x}" cy="${y}" r="1.6" fill="rgba(255,255,255,.75)"/>`)
    .join("");

  return `<svg width="${w}" height="${h}" viewBox="0 0 ${w} ${h}" fill="none" aria-hidden="true">
    <path d="${d}" stroke="rgba(255,255,255,.55)" stroke-width="2" fill="none" stroke-linecap="round"/>
    ${dots}
  </svg>`;
}

export function renderDaily(b: WeatherBundle, s: Settings): HTMLElement {

  const card = document.createElement("section");
  card.className = "card daily";

  const gmaxs = days.map((d) => d.tempMax);
  const lo = Math.min(...gmins);

  const span = hi - lo || 1;

  for (let i = 0; i < days.length; i++) {
    const d = days[i];
    const cond = describe(d.weatherCode, true);
    const row = document.createElement("div");
    row.className = "daily-row";

    const left = ((d.tempMin - lo) / span) * 100;
    const width = ((d.tempMax - d.tempMin) / span) * 100;
    const fill = `background:linear-gradient(90deg, ${tempColor(d.tempMin)}, ${tempColor(d.tempMax)});left:${left}%;width:${width}%`;

      ? `<div class="daily-rain">${d.precipitationProbabilityMax}%</div>`

      <div class="daily-day">${dayName(d.date, i)}</div>

      <div class="daily-min">${temp(d.tempMin, s)}&deg;</div>

    `;

import "./styles.css";
import type { Location, WeatherBundle } from "./types";
import { describe, skyGradient } from "./theme";
import { getCached, getSaved, getWeather, saveLocation } from "./api";
import { state, set } from "./state";
import { renderHero } from "./ui/hero";
import { renderHourly } from "./ui/hourly";
import { renderDaily } from "./ui/daily";

const DEFAULT: Location = {
  name: "Berlin",
  country: "Germany",
  admin1: "Berlin",
  latitude: 52.52,
  longitude: 13.405,
};

let skyFront = 0;
let skyEls: HTMLDivElement[] = [];

function setupShell(): { main: HTMLElement } {
  const app = document.getElementById("app");
  if (!app) throw new Error("missing #app");

  const sky = document.createElement("div");
  sky.className = "sky";
  skyEls = [0, 1].map(() => {
    const l = document.createElement("div");
    l.className = "sky-layer";
    sky.appendChild(l);
    return l;
  });

  const sidebar = document.createElement("aside");
  sidebar.id = "sidebar";
  sidebar.className = "sidebar";

  const main = document.createElement("main");
  main.id = "main";

  app.replaceChildren(sky, sidebar, main);
  return { main };
}

function applySky(bundle: WeatherBundle): void {
  const c = bundle.current;
  const cond = describe(c.weatherCode, c.isDay);
  const grad = skyGradient(cond.kind, c.isDay);
  const back = skyEls[1 - skyFront];
  const front = skyEls[skyFront];
  if (front.style.backgroundImage === grad) return;

  back.style.backgroundImage = grad;
  // force reflow so the transition triggers
  void back.offsetWidth;
  back.classList.add("active");
  front.classList.remove("active");
  skyFront = 1 - skyFront;
}

function renderMain(main: HTMLElement, b: WeatherBundle): void {
  main.replaceChildren(
    renderHero(b, state.settings),
    renderHourly(b, state.settings),
  );
  const grid = document.createElement("section");
  grid.id = "grid";
  grid.className = "grid";
  const daily = renderDaily(b, state.settings);
  daily.classList.add("span-2", "row-2");
  grid.appendChild(daily);
  main.appendChild(grid);
}

function renderPlaceholder(main: HTMLElement, text: string): void {
  const box = document.createElement("div");
  box.className = "loading";
  box.textContent = text;
  main.replaceChildren(box);
}

function renderError(main: HTMLElement, retry: () => void): void {
  const box = document.createElement("div");
  box.className = "error";
  const msg = document.createElement("div");
  msg.textContent = state.error ?? "Something went wrong.";
  const btn = document.createElement("button");
  btn.textContent = "Retry";
  btn.addEventListener("click", retry);
  box.append(msg, btn);
  main.replaceChildren(box);
}

async function load(loc: Location, main: HTMLElement): Promise<void> {
  set({ current: loc, loading: true, error: null });
  if (!state.bundle) renderPlaceholder(main, "Loading weather\u2026");

  try {
    const b = await getWeather(loc, false);
    set({ bundle: b, loading: false });
    applySky(b);
    renderMain(main, b);
  } catch (err) {
    set({ loading: false, error: String(err) });
    renderError(main, () => void load(loc, main));
  }
}

async function init(): Promise<void> {
  const { main } = setupShell();

  let saved: Location[] = [];
  try {
    saved = await getSaved();
  } catch {
    saved = [];
  }
  if (saved.length === 0) {
    try {
      saved = await saveLocation(DEFAULT);
    } catch {
      saved = [DEFAULT];
    }
  }
  set({ saved });
  const loc = saved[0] ?? DEFAULT;

  // Show cached data instantly if present.
  try {
    const cached = await getCached(loc);
    if (cached) {
      set({ bundle: cached, current: loc });
      applySky(cached);
      renderMain(main, cached);
    }
  } catch {
    /* no cache */
  }

  await load(loc, main);

  const refresh = () => {
    const cur = state.current;
    if (!cur || state.loading) return;
    void (async () => {
      try {
        const b = await getWeather(cur, false);
        set({ bundle: b });
        applySky(b);
        renderMain(main, b);
      } catch {
        /* keep last bundle */
      }
    })();
  };

  setInterval(refresh, 10 * 60 * 1000);
  document.addEventListener("visibilitychange", () => {
    if (document.visibilityState === "visible") refresh();
  });
}

void init();
