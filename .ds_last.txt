let root: HTMLDivElement | null = null;
let lastKey = "";

/** Deterministic 0..1 generator (mulberry32) so positions never churn between renders. */
function rng(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function stars(): HTMLDivElement {
  const el = document.createElement("div");
  el.className = "amb-stars";
  const rand = rng(1337);
  const group = (cls: string, n: number): string => {
    const shadows: string[] = [];
    for (let i = 0; i < n; i++) {
      const x = (rand() * 100).toFixed(2);
      const y = (rand() * 70).toFixed(2);
      shadows.push(`${x}vw ${y}vh 0 0 rgba(255,255,255,${(0.4 + rand() * 0.4).toFixed(2)})`);
    }
    const g = document.createElement("div");
    g.className = cls;
    g.style.boxShadow = shadows.join(", ");
    el.appendChild(g);
    return cls;
  };
  group("amb-stars-a", 20);
  group("amb-stars-b", 20);
  return el;
}

function clouds(count: number): HTMLDivElement {
  const el = document.createElement("div");
  el.className = "amb-clouds";
  for (let i = 0; i < count; i++) {
    const c = document.createElement("div");
    c.className = "amb-cloud";
    c.style.top = `${5 + i * (80 / count) + (i % 2 ? 4 : 0)}%`;
    c.style.animationDuration = `${90 + i * 18}s`;
    c.style.animationDelay = `${-i * 7}s`;
    el.appendChild(c);
  }
  return el;
}

function bands(): HTMLDivElement {
  const el = document.createElement("div");
  el.className = "amb-fog";
  for (let i = 0; i < 3; i++) {
    const b = document.createElement("div");
    b.className = i % 2 ? "amb-band alt" : "amb-band";
    b.style.top = `${18 + i * 26}%`;
    b.style.animationDuration = `${70 + i * 25}s`;
    el.appendChild(b);
  }
  return el;
}

function rain(kind: Kind): HTMLDivElement {
  const el = document.createElement("div");
  el.className = kind === "storm" ? "amb-rain storm" : `amb-rain ${kind}`;
  for (let i = 0; i < 2; i++) {
    const layer = document.createElement("div");
    layer.className = i ? "amb-rain-layer far" : "amb-rain-layer";
    el.appendChild(layer);
  }
  return el;
}

function snow(): HTMLDivElement {
  const el = document.createElement("div");
  el.className = "amb-snow";
  for (let i = 0; i < 2; i++) {
    const layer = document.createElement("div");
    layer.className = i ? "amb-snow-layer big" : "amb-snow-layer";
    el.appendChild(layer);
  }
  return el;
}

function build(kind: Kind, isDay: boolean): HTMLDivElement {
  const wrap = document.createElement("div");
  wrap.className = `amb-layer amb-${kind}${isDay ? " day" : " night"}`;
  switch (kind) {
    case "clear":
      wrap.appendChild(isDay ? Object.assign(document.createElement("div"), { className: "amb-glow" }) : stars());
      break;
    case "partly":
      wrap.appendChild(clouds(3));
      break;
    case "cloudy":
      wrap.appendChild(clouds(5));
      break;
    case "fog":
      wrap.appendChild(bands());
      break;
    case "drizzle":
    case "rain":
      wrap.appendChild(rain(kind));
      break;
    case "snow":
      wrap.appendChild(snow());
      break;
    case "storm":
      wrap.appendChild(rain(kind));
      wrap.appendChild(Object.assign(document.createElement("div"), { className: "amb-flash" }));
      break;
  }
  return wrap;
}

export function setAmbient(kind: Kind, isDay: boolean): void {
  const key = `${kind}-${isDay}`;
  if (root && key === lastKey) return;
  lastKey = key;

  if (!root) {
    root = document.createElement("div");
    root.className = "ambient";
    root.setAttribute("aria-hidden", "true");
    const sky = document.querySelector(".sky");
    if (sky?.parentNode === document.body) sky.after(root);
    else document.body.appendChild(root);
    if (document.visibilityState === "hidden") root.classList.add("paused");
    document.addEventListener("visibilitychange", () => {
      root?.classList.toggle("paused", document.visibilityState === "hidden");
    });
  }

  const next = build(kind, isDay);
  const old = root.lastElementChild;
  root.appendChild(next);
  if (!old) return;
  old.classList.add("out");
  const done = () => old.remove();
  old.addEventListener("transitionend", done, { once: true });
  window.setTimeout(done, 700);
}

import "./styles.css";
import "./styles/cards.css";
import "./styles/sidebar.css";
import type { Location, WeatherBundle } from "./types";
import { describe, skyGradient } from "./theme";
import { getCached, getSaved, getWeather, saveLocation } from "./api";
import { state, set, subscribe } from "./state";
import { renderHero } from "./ui/hero";
import { renderHourly } from "./ui/hourly";
import { renderDaily } from "./ui/daily";
import { renderCards } from "./ui/cards";
import { initSidebar } from "./ui/sidebar";
import { setAmbient } from "./ambient";

const DEFAULT: Location = {
  name: "Berlin",
  country: "Germany",
  admin1: "Berlin",
  latitude: 52.52,
  longitude: 13.405,
};

let skyFront = 0;
let skyEls: HTMLDivElement[] = [];
let currentMain: HTMLElement | null = null;

function setupShell(): { main: HTMLElement } {
  const app = document.getElementById("app");
  if (!app) throw new Error("missing #app");

  const sky = document.createElement("div");
  sky.className = "sky";
  skyEls = [0, 1].map(() => {
    const l = document.createElement("div");
    l.className = "sky-layer";
    sky.appendChild(l);
    return l;
  });

  const sidebar = document.createElement("aside");
  sidebar.id = "sidebar";
  sidebar.className = "sidebar";

  const main = document.createElement("main");
  main.id = "main";

  app.replaceChildren(sky, sidebar, main);
  return { main };
}

function applySky(bundle: WeatherBundle): void {
  const c = bundle.current;
  const cond = describe(c.weatherCode, c.isDay);
  const grad = skyGradient(cond.kind, c.isDay);
  const back = skyEls[1 - skyFront];
  const front = skyEls[skyFront];
  if (front.style.backgroundImage === grad) return;

  back.style.backgroundImage = grad;
  void back.offsetWidth;
  back.classList.add("active");
  front.classList.remove("active");
  skyFront = 1 - skyFront;
  setAmbient(cond.kind, c.isDay);
}

function renderMain(main: HTMLElement, b: WeatherBundle): void {
  currentMain = main;
  main.replaceChildren(
    renderHero(b, state.settings),
    renderHourly(b, state.settings),
  );
  const grid = document.createElement("section");
  grid.id = "grid";
  grid.className = "grid";
  const daily = renderDaily(b, state.settings);
  daily.classList.add("span-2", "row-2");
  grid.appendChild(daily);
  for (const card of renderCards(b, state.settings)) grid.appendChild(card);
  main.appendChild(grid);
}

function renderPlaceholder(main: HTMLElement, text: string): void {
  const box = document.createElement("div");
  box.className = "loading";
  box.textContent = text;
  main.replaceChildren(box);
}

function renderError(main: HTMLElement, retry: () => void): void {
  const box = document.createElement("div");
  box.className = "error";
  const msg = document.createElement("div");
  msg.textContent = state.error ?? "Something went wrong.";
  const btn = document.createElement("button");
  btn.textContent = "Retry";
  btn.addEventListener("click", retry);
  box.append(msg, btn);
  main.replaceChildren(box);
}

async function selectLocation(loc: Location): Promise<void> {
  const main = currentMain ?? document.getElementById("main");
  if (!main) return;

  set({ current: loc, loading: true, error: null });

  try {
    const cached = await getCached(loc);
    if (cached) {
      set({ bundle: cached });
      applySky(cached);
      renderMain(main, cached);
    } else if (!state.bundle) {
      renderPlaceholder(main, "Loading weather\u2026");
    }
  } catch {
    /* no cache */
  }

  try {
    const b = await getWeather(loc, false);
    set({ bundle: b, loading: false });
    applySky(b);
    renderMain(main, b);
  } catch (err) {
    set({ loading: false, error: String(err) });
    renderError(main, () => void selectLocation(loc));
  }
}

async function init(): Promise<void> {
  const { main } = setupShell();
  currentMain = main;

  initSidebar(document.getElementById("sidebar")!, (loc) => {
    void selectLocation(loc);
  });

  let saved: Location[] = [];
  try {
    saved = await getSaved();
  } catch {
    saved = [];
  }
  if (saved.length === 0) {
    try {
      saved = await saveLocation(DEFAULT);
    } catch {
      saved = [DEFAULT];
    }
  }
  set({ saved });
  const loc = saved[0] ?? DEFAULT;

  try {
    const cached = await getCached(loc);
    if (cached) {
      set({ bundle: cached, current: loc });
      applySky(cached);
      renderMain(main, cached);
    }
  } catch {
    /* no cache */
  }

  set({ current: loc, loading: true, error: null });
  try {
    const b = await getWeather(loc, false);
    set({ bundle: b, loading: false });
    applySky(b);
    renderMain(main, b);
  } catch (err) {
    set({ loading: false, error: String(err) });
    renderError(main, () => void selectLocation(loc));
  }

  const refresh = () => {
    const cur = state.current;
    if (!cur || state.loading) return;
    void (async () => {
      try {
        const b = await getWeather(cur, false);
        set({ bundle: b });
        applySky(b);
        renderMain(main, b);
      } catch {
        /* keep last bundle */
      }
    })();
  };

  // Re-render on unit changes without refetching.
  subscribe((s) => {
    if (s.bundle && s.settings) {
      const m = currentMain;
      if (m && m.querySelector("#grid")) renderMain(m, s.bundle);
    }
  });

  setInterval(refresh, 10 * 60 * 1000);
  document.addEventListener("visibilitychange", () => {
    if (document.visibilityState === "visible") refresh();
  });
}

void init();
