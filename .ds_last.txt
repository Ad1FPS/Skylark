const SAVED: &str = "saved.json";

pub fn cache_key(loc: &Location) -> String {
    format!("{:.2}_{:.2}.json", loc.latitude, loc.longitude)
}

pub fn read_cache(dir: &Path, loc: &Location) -> Option<WeatherBundle> {
    let text = fs::read_to_string(dir.join(cache_key(loc))).ok()?;
    serde_json::from_str(&text).ok()
}

pub fn write_cache(dir: &Path, bundle: &WeatherBundle) {
    let _ = fs::create_dir_all(dir);
    let Ok(text) = serde_json::to_string(bundle) else {
        return;
    };
    let key = cache_key(&bundle.location);
    let tmp = dir.join(format!("{key}.tmp"));
    if fs::write(&tmp, text).is_ok() {
        let _ = fs::rename(&tmp, dir.join(key));
    }
}

pub fn read_saved(dir: &Path) -> Vec<Location> {
    fs::read_to_string(dir.join(SAVED))
        .ok()
        .and_then(|t| serde_json::from_str(&t).ok())
        .unwrap_or_default()
}

pub fn write_saved(dir: &Path, list: &[Location]) -> Result<(), String> {
    fs::create_dir_all(dir).map_err(|e| e.to_string())?;
    let text = serde_json::to_string(list).map_err(|e| e.to_string())?;
    fs::write(dir.join(SAVED), text).map_err(|e| e.to_string())
}

pub fn parse_forecast(json: &str, loc: &Location, air: Option<Air>, now: i64) -> Result<WeatherBundle, String> {
    let raw: RawForecast = serde_json::from_str(json).map_err(|e| format!("bad forecast: {e}"))?;

    let rc = raw.current.ok_or("missing current")?;
    let current = Current {
        time: rc.time,
        temperature: rc.temperature_2m.unwrap_or(0.0),
        apparent: rc.apparent_temperature.unwrap_or(0.0),
        humidity: rc.relative_humidity_2m.unwrap_or(0.0).clamp(0.0, 255.0) as u8,
        is_day: rc.is_day.unwrap_or(0) != 0,
        weather_code: rc.weather_code.unwrap_or(0),
        wind_speed: rc.wind_speed_10m.unwrap_or(0.0),
        wind_direction: rc.wind_direction_10m.unwrap_or(0.0).clamp(0.0, u16::MAX as f32) as u16,
        pressure: rc.pressure_msl.unwrap_or(0.0),
        visibility: rc.visibility.unwrap_or(0.0),
        uv_index: rc.uv_index.unwrap_or(0.0),
        precipitation: rc.precipitation.unwrap_or(0.0),
    };

    let mut hourly = Vec::new();
    if let Some(h) = &raw.hourly {
        let start_key = first_key(&current.time);
        let start = h
            .time
            .iter()
            .position(|t| first_key(t) == start_key)
            .unwrap_or(0);
        for i in start..h.time.len().min(start + 24) {
            hourly.push(Hour {
                time: h.time[i].clone(),
                temperature: idx(&h.temperature_2m, i),
                weather_code: idx(&h.weather_code, i),
                precipitation_probability: h.precipitation_probability.get(i).copied().flatten(),
                is_day: idx(&h.is_day, i) != 0,
            });
        }
    }

    let mut daily = Vec::new();
    if let Some(d) = &raw.daily {
        for i in 0..d.time.len().min(10) {
            daily.push(Day {
                date: d.time[i].clone(),
                weather_code: idx(&d.weather_code, i),
                temp_max: idx(&d.temperature_2m_max, i),
                temp_min: idx(&d.temperature_2m_min, i),
                precipitation_probability_max: d.precipitation_probability_max.get(i).copied().flatten(),
                precipitation_sum: idx(&d.precipitation_sum, i),
                uv_index_max: idx(&d.uv_index_max, i),
                sunrise: d.sunrise.get(i).cloned().unwrap_or_default(),
                sunset: d.sunset.get(i).cloned().unwrap_or_default(),
            });
        }
    }

    let mut minutely = Vec::new();
    if let Some(m) = &raw.minutely_15 {
        for (i, time) in m.time.iter().enumerate().take(32) {
            minutely.push(Minute {
                time: time.clone(),
                precipitation: m.precipitation.get(i).copied().flatten().unwrap_or(0.0),
            });
        }
    }

    Ok(WeatherBundle {
        location: loc.clone(),
        fetched_at: now,
        stale: false,
        timezone: raw.timezone,
        utc_offset_seconds: raw.utc_offset_seconds,
        current,
        hourly,
        daily,
        air,
        minutely,
    })
}

pub fn parse_geocode(json: &str) -> Vec<Location> {

pub fn parse_ip_location(json: &str) -> Result<Location, String> {
    let raw: RawIp =
        serde_json::from_str(json).map_err(|_| "Could not determine location".to_string())?;
    if !raw.success || raw.city.trim().is_empty() {
        return Err("Could not determine location".into());
    }
    let admin1 = if raw.region.trim().is_empty() {
        None
    } else {
        Some(raw.region)
    };
    Ok(Location {
        name: raw.city,
        country: raw.country,
        admin1,
        latitude: raw.latitude,
        longitude: raw.longitude,
    })
}

/// First 13 chars of an ISO time: `YYYY-MM-DDTHH`.
fn first_key(time: &str) -> &str {
    time.get(..13).unwrap_or(time)
}

/// Builds a query string, percent-encoding values. `reqwest` without the
/// `query` feature lacks `RequestBuilder::query`, so we assemble URLs directly.
fn build_url(base: &str, params: &[(&str, &str)]) -> String {
    let mut out = String::from(base);
    for (i, (k, v)) in params.iter().enumerate() {
        out.push(if i == 0 { '?' } else { '&' });
        out.push_str(&encode(k));
        out.push('=');
        out.push_str(&encode(v));
    }
    out
}

fn encode(raw: &str) -> String {
    let mut out = String::with_capacity(raw.len());
    for b in raw.bytes() {
        match b {
            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'_' | b'.' | b'~' => {
                out.push(b as char)
            }
            _ => out.push_str(&format!("%{b:02X}")),
        }
    }
    out
}

async fn get_text(resp: reqwest::Response) -> Result<String, String> {
    let resp = resp.error_for_status().map_err(|e| e.to_string())?;
    resp.text().await.map_err(|e| e.to_string())
}

pub async fn fetch_bundle(client: &reqwest::Client, loc: &Location) -> Result<WeatherBundle, String> {
    let lat = loc.latitude.to_string();
    let lon = loc.longitude.to_string();

    let forecast_url = build_url(
        FORECAST_URL,
        &[
            ("latitude", &lat),
            ("longitude", &lon),
            ("current", CURRENT_FIELDS),
            ("hourly", HOURLY_FIELDS),
            ("daily", DAILY_FIELDS),
            ("minutely_15", "precipitation"),
            ("forecast_minutely_15", "32"),
            ("past_minutely_15", "0"),
            ("forecast_days", "10"),
            ("timezone", "auto"),
        ],
    );
    let air_url = build_url(
        AIR_URL,
        &[
            ("latitude", &lat),
            ("longitude", &lon),
            ("current", "us_aqi,pm2_5"),
        ],
    );

    let (forecast_res, air_res) = tokio::join!(client.get(&forecast_url).send(), client.get(&air_url).send());

    let forecast_json = get_text(forecast_res.map_err(|e| e.to_string())?).await?;
    let air = match air_res {
        Ok(resp) => match get_text(resp).await {
            Ok(text) => parse_air(&text),
            Err(_) => None,
        },
        Err(_) => None,
    };

    parse_forecast(&forecast_json, loc, air, now_secs())
}

}

    if query.chars().count() < 2 {
        return Ok(Vec::new());
    }
    weather::search(&http.0, &query).await
}

#[tauri::command]
pub async fn locate_ip(http: State<'_, Http>) -> Result<Location, String> {
    weather::locate_ip(&http.0).await
}

#[tauri::command]
pub async fn get_cached(app: AppHandle, location: Location) -> Option<WeatherBundle> {
    let dir = cache_dir(&app).ok()?;
    store::read_cache(&dir, &location)
}

#[tauri::command]
pub async fn get_weather(
    app: AppHandle,
    http: State<'_, Http>,
    location: Location,
    force: bool,
) -> Result<WeatherBundle, String> {
    let cached = get_cached(app.clone(), location.clone()).await;

    if !force
        && let Some(bundle) = &cached
        && now_secs() - bundle.fetched_at < CACHE_TTL_SECS
    {
        return Ok(bundle.clone());
    }

    match weather::fetch_bundle(&http.0, &location).await {
        Ok(bundle) => {
            if let Ok(dir) = cache_dir(&app) {
                store::write_cache(&dir, &bundle);
            }
            Ok(bundle)
        }
        Err(e) => match cached {
            Some(mut bundle) => {
                bundle.stale = true;
                Ok(bundle)
            }
            None => Err(e),
        },
    }
}

#[tauri::command]
pub fn get_saved(app: AppHandle) -> Vec<Location> {
    data_dir(&app)
        .map(|dir| store::read_saved(&dir))
        .unwrap_or_default()
}

fn same_place(a: &Location, lat: f64, lon: f64) -> bool {
    (a.latitude * 100.0).round() == (lat * 100.0).round()
        && (a.longitude * 100.0).round() == (lon * 100.0).round()
}

#[tauri::command]
pub fn save_location(app: AppHandle, location: Location) -> Result<Vec<Location>, String> {
    let dir = data_dir(&app)?;
    let mut list = store::read_saved(&dir);
    if !list
        .iter()
        .any(|l| same_place(l, location.latitude, location.longitude))
    {
        list.push(location);
        store::write_saved(&dir, &list)?;
    }
    Ok(list)
}

#[tauri::command]
pub fn remove_location(
    app: AppHandle,
    latitude: f64,
    longitude: f64,
) -> Result<Vec<Location>, String> {
    let dir = data_dir(&app)?;
    let mut list = store::read_saved(&dir);
    list.retain(|l| !same_place(l, latitude, longitude));
    store::write_saved(&dir, &list)?;
    Ok(list)
}

#[tauri::command]
pub fn set_current_location(app: AppHandle, location: Location) -> Result<(), String> {
    let dir = data_dir(&app)?;
    std::fs::create_dir_all(&dir).map_err(|e| e.to_string())?;
    let text = serde_json::to_string(&location).map_err(|e| e.to_string())?;
    std::fs::write(dir.join(CURRENT_FILE), text).map_err(|e| e.to_string())?;
    tray::refresh_now(&app);
    Ok(())
}

pub fn read_current_location(app: &AppHandle) -> Option<Location> {
    let dir = data_dir(app).ok()?;
    if let Ok(text) = std::fs::read_to_string(dir.join(CURRENT_FILE))
        && let Ok(loc) = serde_json::from_str::<Location>(&text)
    {
        return Some(loc);
    }
    store::read_saved(&dir).into_iter().next()
}

use std::time::Duration;

use tauri::menu::{Menu, MenuEvent, MenuItem, PredefinedMenuItem};
use tauri::tray::{MouseButton, MouseButtonState, TrayIconBuilder, TrayIconEvent};
use tauri::{AppHandle, Manager, WebviewWindowBuilder};

use crate::commands::{self, Http};
use crate::store;
use crate::weather;

pub struct TrayStatus(pub MenuItem<tauri::Wry>);

const STATUS_INITIAL: &str = "Weather";

pub fn setup(app: &AppHandle) -> tauri::Result<()> {
    let status = MenuItem::with_id(app, "status", STATUS_INITIAL, false, None::<&str>)?;
    let show = MenuItem::with_id(app, "show", "Show", true, None::<&str>)?;
    let refresh = MenuItem::with_id(app, "refresh", "Refresh", true, None::<&str>)?;
    let sep = PredefinedMenuItem::separator(app)?;
    let quit = MenuItem::with_id(app, "quit", "Quit", true, None::<&str>)?;
    let menu = Menu::with_items(app, &[&status, &show, &refresh, &sep, &quit])?;

    let mut builder = TrayIconBuilder::with_id("main")
        .tooltip(STATUS_INITIAL)
        .menu(&menu)
        .show_menu_on_left_click(false)
        .on_menu_event(|app, event: MenuEvent| match event.id().as_ref() {
            "show" => show_window(app),
            "refresh" => refresh_now(app),
            "quit" => app.exit(0),
            _ => {}
        })
        .on_tray_icon_event(|tray, event| {
            if let TrayIconEvent::Click {
                button: MouseButton::Left,
                button_state: MouseButtonState::Up,
                ..
            } = event
            {
                show_window(tray.app_handle());
            }
        });
    if let Some(icon) = app.default_window_icon().cloned() {
        builder = builder.icon(icon);
    }
    builder.build(app)?;

    app.manage(TrayStatus(status));
    Ok(())
}

pub fn show_window(app: &AppHandle) {
    if let Some(window) = app.get_webview_window("main") {
        let _ = window.show();
        let _ = window.unminimize();
        let _ = window.set_focus();
        return;
    }
    if let Some(cfg) = app.config().app.windows.first()
        && let Ok(window) = WebviewWindowBuilder::from_config(app, cfg).and_then(|b| b.build())
    {
        let _ = window.set_focus();
    }
}

pub fn start_background_refresh(app: &AppHandle) {
    let handle = app.clone();
    std::thread::spawn(move || {
        std::thread::sleep(Duration::from_secs(5));
        loop {
            let writer = handle.clone();
            tauri::async_runtime::spawn(async move { refresh(&writer).await });
            std::thread::sleep(Duration::from_secs(15 * 60));
        }
    });
}

pub fn refresh_now(app: &AppHandle) {
    let handle = app.clone();
    tauri::async_runtime::spawn(async move { refresh(&handle).await });
}

async fn refresh(app: &AppHandle) {
    let Some(location) = commands::read_current_location(app) else {
        return;
    };
    let Some(http) = app.try_state::<Http>() else {
        return;
    };
    let Ok(bundle) = weather::fetch_bundle(&http.0, &location).await else {
        return;
    };
    if let Ok(dir) = commands::cache_dir(app) {
        store::write_cache(&dir, &bundle);
    }
    let minutes = minutes_to_rain(&bundle.minutely);
    let text = summary(
        &bundle.location.name,
        bundle.current.temperature,
        bundle.current.weather_code,
        minutes,
    );
    apply_status(app, &text);
}

fn apply_status(app: &AppHandle, text: &str) {
    if let Some(tray) = app.tray_by_id("main") {
        let _ = tray.set_tooltip(Some(text));
    }
    if let Some(state) = app.try_state::<TrayStatus>() {
        let _ = state.0.set_text(text);
    }
}

fn minutes_to_rain(minutely: &[weather::Minute]) -> Option<i64> {
    let first = minutely.first()?;
    let start = parse_minutes(&first.time)?;
    let pos = minutely.iter().position(|m| m.precipitation >= 0.1)?;
    let at = parse_minutes(&minutely[pos].time)?;
    Some((at - start).max(0))
}

fn parse_minutes(time: &str) -> Option<i64> {
    let hour = time.get(11..13)?.parse::<i64>().ok()?;
    let minute = time.get(14..16)?.parse::<i64>().ok()?;
    Some(hour * 60 + minute)
}

fn wmo_label(code: u8) -> &'static str {
    match code {
        0 => "Clear",
        1 | 2 => "Partly cloudy",
        3 => "Overcast",
        45 | 48 => "Fog",
        51..=57 => "Drizzle",
        61..=67 | 80..=82 => "Rain",
        71..=77 | 85 | 86 => "Snow",
        95..=99 => "Thunderstorm",
        _ => "Weather",
    }
}

pub fn summary(city: &str, temp: f32, code: u8, minutes_to_rain: Option<i64>) -> String {
    let mut text = format!("{city} {}° {}", temp.round() as i32, wmo_label(code));
    match minutes_to_rain {
        Some(0) => text.push_str(" · rain now"),
        Some(n) => text.push_str(&format!(" · rain in {n} min")),
        None => {}
    }
    text
}

mod commands;

use commands::Http;

#[cfg_attr(mobile, tauri::mobile_entry_point)]
pub fn run() {
    tauri::Builder::default()
        .plugin(tauri_plugin_opener::init())
        .manage(Http::new())
        .invoke_handler(tauri::generate_handler![
            commands::search_city,
            commands::locate_ip,
            commands::get_cached,
            commands::get_weather,
            commands::get_saved,
            commands::save_location,
            commands::remove_location,
            commands::set_current_location,
        ])
        .setup(|app| {
            tray::setup(app.handle())?;
            tray::start_background_refresh(app.handle());
            Ok(())
        })
        .build(tauri::generate_context!())
        .expect("error while building tauri application")
        .run(|_app, event| {
            if let tauri::RunEvent::ExitRequested { api, code, .. } = event
                && code.is_none()
            {
                api.prevent_exit();
            }
        });
}
